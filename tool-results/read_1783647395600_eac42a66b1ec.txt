    76→/* ─── Typing Text Hook ─── */
    77→function useTypingText(phrases: string[], typingSpeed = 80, deletingSpeed = 40, pauseDuration = 2000) {
    78→  const [displayed, setDisplayed] = useState('')
    79→  const [phraseIndex, setPhraseIndex] = useState(0)
    80→  const [isDeleting, setIsDeleting] = useState(false)
    81→
    82→  useEffect(() => {
    83→    const currentPhrase = phrases[phraseIndex]
    84→    let timeout: NodeJS.Timeout
    85→
    86→    if (!isDeleting && displayed === currentPhrase) {
    87→      timeout = setTimeout(() => setIsDeleting(true), pauseDuration)
    88→    } else if (isDeleting && displayed === '') {
    89→      setIsDeleting(false)
    90→      setPhraseIndex((prev) => (prev + 1) % phrases.length)
    91→    } else {
    92→      timeout = setTimeout(
    93→        () => {
    94→          setDisplayed(
    95→            isDeleting
    96→              ? currentPhrase.substring(0, displayed.length - 1)
    97→              : currentPhrase.substring(0, displayed.length + 1)
    98→          )
    99→        },
   100→        isDeleting ? deletingSpeed : typingSpeed
   101→      )
   102→    }
   103→    return () => clearTimeout(timeout)
   104→  }, [displayed, isDeleting, phraseIndex, phrases, typingSpeed, deletingSpeed, pauseDuration])
   105→
   106→  return displayed
   107→}

   109→/* ─── useLocalStorage Hook ─── */
   110→function useLocalStorage<T>(key: string, initialValue: T): [T, (v: T | ((prev: T) => T)) => void] {
   111→  const [stored, setStored] = useState<T>(() => {
   112→    if (typeof window === 'undefined') return initialValue
   113→    try {
   114→      const item = localStorage.getItem(key)
   115→      return item ? JSON.parse(item) : initialValue
   116→    } catch { return initialValue }
   117→  })
   118→  const setValue = useCallback((value: T | ((prev: T) => T)) => {
   119→    setStored(prev => {
   120→      const next = value instanceof Function ? value(prev) : value
   121→      if (typeof window !== 'undefined') localStorage.setItem(key, JSON.stringify(next))
   122→      return next
   123→    })
   124→  }, [key])
   125→  return [stored, setValue]
   126→}

   128→/* ─── useCountdown Hook ─── */
   129→function useCountdown(targetDate: Date) {
   130→  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 })
   131→  useEffect(() => {
   132→    const calc = () => {
   133→      const diff = targetDate.getTime() - Date.now()
   134→      if (diff <= 0) return { days: 0, hours: 0, minutes: 0, seconds: 0 }
   135→      return {
   136→        days: Math.floor(diff / 86400000),
   137→        hours: Math.floor((diff % 86400000) / 3600000),
   138→        minutes: Math.floor((diff % 3600000) / 60000),
   139→        seconds: Math.floor((diff % 60000) / 1000),
   140→      }
   141→    }
   142→    setTimeLeft(calc())
   143→    const interval = setInterval(() => setTimeLeft(calc()), 1000)
   144→    return () => clearInterval(interval)
   145→  }, [targetDate])
   146→  return timeLeft
   147→}

   149→/* ─── Animated Counter Component ─── */
   150→function AnimatedStat({ end, suffix, children }: { end: number; suffix?: string; children: React.ReactNode }) {
   151→  const [count, setCount] = useState(0)
   152→  const [started, setStarted] = useState(false)
   153→  const ref = useRef<HTMLDivElement>(null)
   154→
   155→  useEffect(() => {
   156→    if (!started) return
   157→    let start = 0
   158→    let frameId: number
   159→    const step = (timestamp: number) => {
   160→      if (!start) start = timestamp
   161→      const progress = Math.min((timestamp - start) / 1500, 1)
   162→      setCount(Math.floor(progress * end))
   163→      if (progress < 1) frameId = requestAnimationFrame(step)
   164→    }
   165→    frameId = requestAnimationFrame(step)
   166→    return () => cancelAnimationFrame(frameId)
   167→  }, [started, end])
   168→
   169→  useEffect(() => {
   170→    const el = ref.current
   171→    if (!el) return
   172→    const observer = new IntersectionObserver(
   173→      ([entry]) => { if (entry.isIntersecting) { setStarted(true); observer.disconnect() } },
   174→      { threshold: 0.3 }
   175→    )
   176→    observer.observe(el)
   177→    return () => observer.disconnect()
   178→  }, [])
   179→
   180→  return (
   181→    <div
   182→      ref={ref}
   183→      className="text-center"
   184→    >
   185→      {children}
   186→      <div className="text-3xl sm:text-4xl font-extrabold text-white mb-1">{count}{suffix}</div>
   187→    </div>
   188→  )
   189→}

   191→/* ─── Before / After Slider ─── */
   192→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   193→  const containerRef = useRef<HTMLDivElement>(null)
   194→  const [sliderPos, setSliderPos] = useState(50)
   195→  const [isDragging, setIsDragging] = useState(false)
   196→
   197→  const updatePosition = (clientX: number) => {
   198→    if (!containerRef.current) return
   199→    const rect = containerRef.current.getBoundingClientRect()
   200→    const x = clientX - rect.left
   201→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   202→    setSliderPos(pct)
   203→  }
   204→
   205→  const handlePointerDown = (e: React.PointerEvent) => {
   206→    setIsDragging(true)
   207→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   208→    updatePosition(e.clientX)
   209→  }
   210→
   211→  const handlePointerMove = (e: React.PointerEvent) => {
   212→    if (!isDragging) return
   213→    updatePosition(e.clientX)
   214→  }
   215→
   216→  const handlePointerUp = () => {
   217→    setIsDragging(false)
   218→  }
   219→
   220→  return (
   221→    <div className="space-y-2">
   222→      <div
   223→        ref={containerRef}
   224→        className="relative h-56 sm:h-64 rounded-xl overflow-hidden cursor-col-resize select-none border-2 border-border shadow-lg"
   225→        onPointerDown={handlePointerDown}
   226→        onPointerMove={handlePointerMove}
   227→        onPointerUp={handlePointerUp}
   228→      >
   229→        <img src={after} alt="Après" className="absolute inset-0 w-full h-full object-cover" draggable={false} />
   230→        <div className="absolute inset-0" style={{ clipPath: `inset(0 ${100 - sliderPos}% 0 0)` }}>
   231→          <img src={before} alt="Avant" className="w-full h-full object-cover" draggable={false} />
   232→        </div>
   233→        <span className="absolute top-3 left-3 z-20 bg-red-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">AVANT</span>
   234→        <span className="absolute top-3 right-3 z-20 bg-emerald-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">APRÈS</span>
   235→        <div className="absolute top-0 bottom-0 z-30 w-0.5 bg-white shadow-[0_0_8px_rgba(0,0,0,0.5)]" style={{ left: `${sliderPos}%` }}>
   236→          <div className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 w-10 h-10 rounded-full bg-white shadow-xl flex items-center justify-center border-2 border-white/80">
   237→            <div className="flex items-center gap-0.5">
   238→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-r-[6px] border-t-transparent border-b-transparent border-r-gray-700" />
   239→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-l-[6px] border-t-transparent border-b-transparent border-l-gray-700" />
   240→            </div>
   241→          </div>
   242→        </div>
   243→        {!isDragging && (
   244→          <div className="absolute inset-0 z-10 flex items-center justify-center opacity-0 hover:opacity-100 transition-opacity bg-black/20">
   245→            <div className="bg-white/90 text-gray-800 text-xs font-semibold px-3 py-1.5 rounded-full shadow">
   246→              Glissez pour comparer
   247→            </div>
   248→          </div>
   249→        )}
   250→      </div>
   251→    </div>
   252→  )
   253→}
