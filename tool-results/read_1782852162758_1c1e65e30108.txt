    62→/* ─── Animation Helpers ─── */
    63→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    64→  const ref = useRef(null)
    65→  const isInView = useInView(ref, { once: true, margin: '-60px' })
    66→  return (
    67→    <motion.div
    68→      ref={ref}
    69→      initial={{ opacity: 0, y: 30 }}
    70→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    71→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
    72→      className={className}
    73→    >
    74→      {children}
    75→    </motion.div>
    76→  )
    77→}

    79→function StaggerContainer({ children, className = '' }: { children: React.ReactNode; className?: string }) {
