    70→    70→/* ─── Animated Counter Hook ─── */
    71→    71→function useCounter(end: number, duration: number = 2000) {
    72→    72→  const [count, setCount] = useState(0)
    73→    73→  const [started, setStarted] = useState(false)
    74→    74→  const ref = useRef<HTMLDivElement>(null)
    75→    75→
    76→    76→  useEffect(() => {
    77→    77→    const el = ref.current
    78→    78→    if (!el) return
    79→    79→    const observer = new IntersectionObserver(
    80→    80→      ([entry]) => {
    81→    81→        if (entry.isIntersecting && !started) {
    82→    82→          setStarted(true)
    83→    83→        }
    84→    84→      },
    85→    85→      { threshold: 0.3 }
    86→    86→    )
    87→    87→    observer.observe(el)
    88→    88→    return () => observer.disconnect()
    89→    89→  }, [started])
    90→    90→
    91→    91→  useEffect(() => {
    92→    92→    if (!started) return
    93→    93→    let start = 0
    94→    94→    let frameId: number
    95→    95→    const step = (timestamp: number) => {
    96→    96→      if (!start) start = timestamp
    97→    97→      const progress = Math.min((timestamp - start) / duration, 1)
    98→    98→      setCount(Math.floor(progress * end))
    99→    99→      if (progress < 1) frameId = requestAnimationFrame(step)
   100→   100→    }
   101→   101→    frameId = requestAnimationFrame(step)
   102→   102→    return () => cancelAnimationFrame(frameId)
   103→   103→  }, [started, end, duration])
   104→   104→
   105→   105→  return { count, ref }
   106→   106→}

   109→   109→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   110→   110→  const containerRef = useRef<HTMLDivElement>(null)
   111→   111→  const [sliderPos, setSliderPos] = useState(50)
   112→   112→  const [isDragging, setIsDragging] = useState(false)
   113→   113→
   114→   114→  const updatePosition = (clientX: number) => {
   115→   115→    if (!containerRef.current) return
   116→   116→    const rect = containerRef.current.getBoundingClientRect()
   117→   117→    const x = clientX - rect.left
   118→   118→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   119→   119→    setSliderPos(pct)
   120→   120→  }
   121→   121→
   122→   122→  const handlePointerDown = (e: React.PointerEvent) => {
   123→   123→    setIsDragging(true)
   124→   124→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   125→   125→    updatePosition(e.clientX)
   126→   126→  }
   127→   127→
   128→   128→  const handlePointerMove = (e: React.PointerEvent) => {
   129→   129→    if (!isDragging) return
   130→   130→    updatePosition(e.clientX)
   131→   131→  }
   132→   132→
   133→   133→  const handlePointerUp = () => {
   134→   134→    setIsDragging(false)
   135→   135→  }
   136→   136→
   137→   137→  return (
   138→   138→    <div className="space-y-2">
   139→   139→      <div
   140→   140→        ref={containerRef}
   141→   141→        className="relative h-56 sm:h-64 rounded-xl overflow-hidden cursor-col-resize select-none border-2 border-border shadow-lg"
   142→   142→        onPointerDown={handlePointerDown}
   143→   143→        onPointerMove={handlePointerMove}
   144→   144→        onPointerUp={handlePointerUp}
   145→   145→      >
   146→   146→        <img src={after} alt="Après" className="absolute inset-0 w-full h-full object-cover" draggable={false} />
   147→   147→        <div className="absolute inset-0" style={{ clipPath: `inset(0 ${100 - sliderPos}% 0 0)` }}>
   148→   148→          <img src={before} alt="Avant" className="w-full h-full object-cover" draggable={false} />
   149→   149→        </div>
   150→   150→        <span className="absolute top-3 left-3 z-20 bg-red-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">AVANT</span>

   153→   153→          <div className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 w-10 h-10 rounded-full bg-white shadow-xl flex items-center justify-center border-2 border-white/80">
