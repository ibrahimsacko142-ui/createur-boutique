    91→/* ─── Animation Helpers ─── */
    92→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    93→  const ref = useRef(null)
    94→  const isInView = useInView(ref, { once: true, margin: '-60px' })
    95→  return (
    96→    <motion.div
    97→      ref={ref}
    98→      initial={{ opacity: 0, y: 30 }}
    99→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   100→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
   101→      className={className}
   102→    >
   103→      {children}
   104→    </motion.div>
   105→  )
   106→}

   108→function StaggerContainer({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   109→  const ref = useRef(null)
   110→  const isInView = useInView(ref, { once: true, margin: '-60px' })
   111→  return (
   112→    <motion.div
   113→      ref={ref}
   114→      initial="hidden"
   115→      animate={isInView ? 'visible' : 'hidden'}
   116→      variants={{ hidden: {}, visible: { transition: { staggerChildren: 0.08 } } }}
   117→      className={className}
   118→    >
   119→      {children}
   120→    </motion.div>
   121→  )
   122→}
