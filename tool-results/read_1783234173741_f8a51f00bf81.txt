   103→      if (!containerRef.current) return
   104→      const rect = containerRef.current.getBoundingClientRect()
   105→      const x = clientX - rect.left
   106→      const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   107→      setSliderPos(pct)
   108→    }
   109→
   110→    const handlePointerDown = (e: React.PointerEvent) => {
   111→      setIsDragging(true)
   112→      ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   113→      updatePosition(e.clientX)
   114→    }
   115→
   116→    const handlePointerMove = (e: React.PointerEvent) => {
   117→      if (!isDragging) return
   118→      updatePosition(e.clientX)
   119→    }
   120→
   121→    const handlePointerUp = () => {
   122→      setIsDragging(false)
   123→    }
