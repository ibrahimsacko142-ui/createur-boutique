    85→/* ─── Typing Text Hook ─── */
    86→function useTypingText(phrases: string[], typingSpeed = 80, deletingSpeed = 40, pauseDuration = 2000) {
    87→  const [displayed, setDisplayed] = useState('')
    88→  const [phraseIndex, setPhraseIndex] = useState(0)
    89→  const [isDeleting, setIsDeleting] = useState(false)
    90→
    91→  useEffect(() => {
    92→    const currentPhrase = phrases[phraseIndex]
    93→    let timeout: NodeJS.Timeout
    94→
    95→    if (!isDeleting && displayed === currentPhrase) {
    96→      timeout = setTimeout(() => setIsDeleting(true), pauseDuration)
    97→    } else if (isDeleting && displayed === '') {
    98→      setIsDeleting(false)
    99→      setPhraseIndex((prev) => (prev + 1) % phrases.length)
   100→    } else {
   101→      timeout = setTimeout(
   102→        () => {
   103→          setDisplayed(
   104→            isDeleting
   105→              ? currentPhrase.substring(0, displayed.length - 1)
   106→              : currentPhrase.substring(0, displayed.length + 1)
   107→          )
   108→        },
   109→        isDeleting ? deletingSpeed : typingSpeed
   110→      )
   111→    }
   112→    return () => clearTimeout(timeout)
   113→  }, [displayed, isDeleting, phraseIndex, phrases, typingSpeed, deletingSpeed, pauseDuration])
   114→
   115→  return displayed
   116→}

   118→/* ─── useLocalStorage Hook ─── */
   119→function useLocalStorage<T>(key: string, initialValue: T): [T, (v: T | ((prev: T) => T)) => void] {
   120→  const [stored, setStored] = useState<T>(() => {
   121→    if (typeof window === 'undefined') return initialValue
   122→    try {
   123→      const item = localStorage.getItem(key)
   124→      return item ? JSON.parse(item) : initialValue
   125→    } catch { return initialValue }
   126→  })
   127→  const setValue = useCallback((value: T | ((prev: T) => T)) => {
   128→    setStored(prev => {
   129→      const next = value instanceof Function ? value(prev) : value
   130→      if (typeof window !== 'undefined') localStorage.setItem(key, JSON.stringify(next))
   131→      return next
   132→    })
   133→  }, [key])
   134→  return [stored, setValue]
   135→}

   137→/* ─── useCountdown Hook ─── */
   138→function useCountdown(targetDate: Date) {
   139→  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 })
   140→  useEffect(() => {
   141→    const calc = () => {
   142→      const diff = targetDate.getTime() - Date.now()
   143→      if (diff <= 0) return { days: 0, hours: 0, minutes: 0, seconds: 0 }
   144→      return {
   145→        days: Math.floor(diff / 86400000),
   146→        hours: Math.floor((diff % 86400000) / 3600000),
   147→        minutes: Math.floor((diff % 3600000) / 60000),
   148→        seconds: Math.floor((diff % 60000) / 1000),
   149→      }
   150→    }
   151→    setTimeLeft(calc())
   152→    const interval = setInterval(() => setTimeLeft(calc()), 1000)
   153→    return () => clearInterval(interval)
   154→  }, [targetDate])
   155→  return timeLeft
   156→}

   158→/* ─── Animated Counter Component ─── */
   159→function AnimatedStat({ end, suffix, children }: { end: number; suffix?: string; children: React.ReactNode }) {
   160→  const [count, setCount] = useState(0)
   161→  const [started, setStarted] = useState(false)
   162→  const ref = useRef<HTMLDivElement>(null)
   163→
   164→  useEffect(() => {
   165→    if (!started) return
   166→    let start = 0
   167→    let frameId: number
   168→    const step = (timestamp: number) => {
   169→      if (!start) start = timestamp
   170→      const progress = Math.min((timestamp - start) / 1500, 1)
   171→      setCount(Math.floor(progress * end))
   172→      if (progress < 1) frameId = requestAnimationFrame(step)
   173→    }
   174→    frameId = requestAnimationFrame(step)
   175→    return () => cancelAnimationFrame(frameId)
   176→  }, [started, end])
   177→
   178→  useEffect(() => {
   179→    const el = ref.current
   180→    if (!el) return
   181→    const observer = new IntersectionObserver(
   182→      ([entry]) => { if (entry.isIntersecting) { setStarted(true); observer.disconnect() } },
   183→      { threshold: 0.3 }
   184→    )
   185→    observer.observe(el)
   186→    return () => observer.disconnect()
   187→  }, [])
   188→
   189→  return (
   190→    <div
   191→      ref={ref}
   192→      className="text-center"
   193→    >
   194→      {children}
   195→      <div className="text-3xl sm:text-4xl font-extrabold text-white mb-1">{count}{suffix}</div>
   196→    </div>
   197→  )
   198→}

   200→/* ─── Before / After Slider ─── */
   201→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   202→  const containerRef = useRef<HTMLDivElement>(null)
   203→  const [sliderPos, setSliderPos] = useState(50)
   204→  const [isDragging, setIsDragging] = useState(false)
   205→
   206→  const updatePosition = (clientX: number) => {
   207→    if (!containerRef.current) return
   208→    const rect = containerRef.current.getBoundingClientRect()
   209→    const x = clientX - rect.left
   210→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   211→    setSliderPos(pct)
   212→  }
   213→
   214→  const handlePointerDown = (e: React.PointerEvent) => {
   215→    setIsDragging(true)
   216→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   217→    updatePosition(e.clientX)
   218→  }
   219→
   220→  const handlePointerMove = (e: React.PointerEvent) => {
   221→    if (!isDragging) return
   222→    updatePosition(e.clientX)
   223→  }
   224→
   225→  const handlePointerUp = () => {
   226→    setIsDragging(false)
   227→  }
   228→
   229→  return (
   230→    <div className="space-y-2">
   231→      <div
   232→        ref={containerRef}
   233→        className="relative h-56 sm:h-64 rounded-xl overflow-hidden cursor-col-resize select-none border-2 border-border shadow-lg"
   234→        onPointerDown={handlePointerDown}
   235→        onPointerMove={handlePointerMove}
   236→        onPointerUp={handlePointerUp}
   237→      >
   238→        <img src={after} alt="Après" className="absolute inset-0 w-full h-full object-cover" draggable={false} />
   239→        <div className="absolute inset-0" style={{ clipPath: `inset(0 ${100 - sliderPos}% 0 0)` }}>
   240→          <img src={before} alt="Avant" className="w-full h-full object-cover" draggable={false} />
   241→        </div>
   242→        <span className="absolute top-3 left-3 z-20 bg-red-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">AVANT</span>
   243→        <span className="absolute top-3 right-3 z-20 bg-emerald-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">APRÈS</span>
   244→        <div className="absolute top-0 bottom-0 z-30 w-0.5 bg-white shadow-[0_0_8px_rgba(0,0,0,0.5)]" style={{ left: `${sliderPos}%` }}>
   245→          <div className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 w-10 h-10 rounded-full bg-white shadow-xl flex items-center justify-center border-2 border-white/80">
   246→            <div className="flex items-center gap-0.5">
   247→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-r-[6px] border-t-transparent border-b-transparent border-r-gray-700" />
   248→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-l-[6px] border-t-transparent border-b-transparent border-l-gray-700" />
   249→            </div>
   250→          </div>
   251→        </div>
   252→        {!isDragging && (
   253→          <div className="absolute inset-0 z-10 flex items-center justify-center opacity-0 hover:opacity-100 transition-opacity bg-black/20">
   254→            <div className="bg-white/90 text-gray-800 text-xs font-semibold px-3 py-1.5 rounded-full shadow">
   255→              Glissez pour comparer
   256→            </div>
   257→          </div>
   258→        )}
   259→      </div>
   260→    </div>
   261→  )
   262→}
