   119→function FadeIn({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
   120→  const ref = useRef(null)
   121→  const isInView = useInView(ref, { once: true, margin: '-60px' })
   122→  return (
   123→    <motion.div
   124→      ref={ref}
   125→      initial={{ opacity: 0, y: 30 }}
   126→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   127→      transition={{ duration: 0.5, delay, ease: 'easeOut' }}
   128→      className={className}
   129→    >
   130→      {children}
   131→    </motion.div>
   132→  )
   133→}

   318→/* ─── Animated Counter ─── */
   319→function Counter({ target, duration = 1500, suffix = '', prefix = '' }: { target: number; duration?: number; suffix?: string; prefix?: string }) {
   320→  const [count, setCount] = useState(0)
   321→  const elementRef = useRef<HTMLSpanElement>(null)
   322→  const hasAnimated = useRef(false)
   323→
   324→  useEffect(() => {
   325→    const observer = new IntersectionObserver(
   326→      ([entry]) => {
   327→        if (entry.isIntersecting && !hasAnimated.current) {
   328→          hasAnimated.current = true
   329→          let startTime: number | null = null
   330→          const startValue = 0
   331→
   332→          const animate = (timestamp: number) => {
   333→            if (!startTime) startTime = timestamp
   334→            const progress = Math.min((timestamp - startTime) / duration, 1)
   335→            const easeProgress = progress * (2 - progress)
   336→            const currentValue = Math.floor(easeProgress * (target - startValue) + startValue)
   337→            setCount(currentValue)
   338→
   339→            if (progress < 1) {
   340→              requestAnimationFrame(animate)
   341→            }
   342→          }
   343→
   344→          requestAnimationFrame(animate)
   345→        }
   346→      },
   347→      { threshold: 0.2 }
   348→    )
   349→
   350→    if (elementRef.current) {
   351→      observer.observe(elementRef.current)
   352→    }
   353→
   354→    return () => observer.disconnect()
   355→  }, [target, duration])
   356→
   357→  return (
   358→    <span ref={elementRef} className="tabular-nums">
   359→      {prefix}{count.toLocaleString('fr-FR')}{suffix}
   360→    </span>
   361→  )
   362→}
