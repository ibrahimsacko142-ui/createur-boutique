   150→   349→/* ─── Animated Counter ─── */
   151→   350→function Counter({ target, duration = 1500, suffix = '', prefix = '' }: { target: number; duration?: number; suffix?: string; prefix?: string }) {
   152→   351→  const [count, setCount] = useState(0)
   153→   352→  const elementRef = useRef<HTMLSpanElement>(null)
   154→   353→  const hasAnimated = useRef(false)
   155→   354→
   156→   355→  useEffect(() => {
   157→   356→    const observer = new IntersectionObserver(
   158→   357→      ([entry]) => {
   159→   358→        if (entry.isIntersecting && !hasAnimated.current) {
   160→   359→          hasAnimated.current = true
   161→   360→          let startTime: number | null = null
   162→   361→          const startValue = 0
   163→   362→
   164→   363→          const animate = (timestamp: number) => {
   165→   364→            if (!startTime) startTime = timestamp
   166→   365→            const progress = Math.min((timestamp - startTime) / duration, 1)
   167→   366→            const easeProgress = progress * (2 - progress)
   168→   367→            const currentValue = Math.floor(easeProgress * (target - startValue) + startValue)
   169→   368→            setCount(currentValue)
   170→   369→
   171→   370→            if (progress < 1) {
   172→   371→              requestAnimationFrame(animate)
   173→   372→            }
   174→   373→          }
   175→   374→
   176→   375→          requestAnimationFrame(animate)
   177→   376→        }
   178→   377→      },
   179→   378→      { threshold: 0.2 }
   180→   379→    )
   181→   380→
   182→   381→    if (elementRef.current) {
   183→   382→      observer.observe(elementRef.current)
   184→   383→    }
   185→   384→
   186→   385→    return () => observer.disconnect()
   187→   386→  }, [target, duration])
   188→   387→
   189→   388→  return (
   190→   389→    <span ref={elementRef} className="tabular-nums">
   191→   390→      {prefix}{count.toLocaleString('fr-FR')}{suffix}
   192→   391→    </span>
   193→   392→  )
   194→   393→}
