    71→    71→    71→function useCounter(end: number, duration: number = 2000) {
    72→    72→    72→  const [count, setCount] = useState(0)
    73→    73→    73→  const [started, setStarted] = useState(false)
    74→    74→    74→  const ref = useRef<HTMLDivElement>(null)
    75→    75→    75→
    76→    76→    76→  useEffect(() => {
    77→    77→    77→    const el = ref.current
    78→    78→    78→    if (!el) return
    79→    79→    79→    const observer = new IntersectionObserver(
    80→    80→    80→      ([entry]) => {
    81→    81→    81→        if (entry.isIntersecting && !started) {
    82→    82→    82→          setStarted(true)
    83→    83→    83→        }
    84→    84→    84→      },
    85→    85→    85→      { threshold: 0.3 }
    86→    86→    86→    )
    87→    87→    87→    observer.observe(el)
    88→    88→    88→    return () => observer.disconnect()
    89→    89→    89→  }, [started])
    90→    90→    90→
    91→    91→    91→  useEffect(() => {
    92→    92→    92→    if (!started) return
    93→    93→    93→    let start = 0
    94→    94→    94→    let frameId: number
    95→    95→    95→    const step = (timestamp: number) => {
    96→    96→    96→      if (!start) start = timestamp
    97→    97→    97→      const progress = Math.min((timestamp - start) / duration, 1)
    98→    98→    98→      setCount(Math.floor(progress * end))
    99→    99→    99→      if (progress < 1) frameId = requestAnimationFrame(step)
   100→   100→   100→    }
   101→   101→   101→    frameId = requestAnimationFrame(step)
   102→   102→   102→    return () => cancelAnimationFrame(frameId)
   103→   103→   103→  }, [started, end, duration])
   104→   104→   104→
   105→   105→   105→  return { count, ref }
   106→   106→   106→}

   109→   109→   109→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   110→   110→   110→  const containerRef = useRef<HTMLDivElement>(null)
   111→   111→   111→  const [sliderPos, setSliderPos] = useState(50)
   112→   112→   112→  const [isDragging, setIsDragging] = useState(false)
   113→   113→   113→
   114→   114→   114→  const updatePosition = (clientX: number) => {
   115→   115→   115→    if (!containerRef.current) return
   116→   116→   116→    const rect = containerRef.current.getBoundingClientRect()
   117→   117→   117→    const x = clientX - rect.left
   118→   118→   118→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   119→   119→   119→    setSliderPos(pct)
   120→   120→   120→  }
   121→   121→   121→

   123→   123→   123→    setIsDragging(true)

   128→   128→   128→  const handlePointerMove = (e: React.PointerEvent) => {
   129→   129→   129→    if (!isDragging) return
   130→   130→   130→    updatePosition(e.clientX)
   131→   131→   131→  }

   133→   133→   133→  const handlePointerUp = () => {
