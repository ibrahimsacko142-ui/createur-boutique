    73→/* ─── Typing Text Hook ─── */
    74→function useTypingText(phrases: string[], typingSpeed = 80, deletingSpeed = 40, pauseDuration = 2000) {
    75→  const [displayed, setDisplayed] = useState('')
    76→  const [phraseIndex, setPhraseIndex] = useState(0)
    77→  const [isDeleting, setIsDeleting] = useState(false)
    78→
    79→  useEffect(() => {
    80→    const currentPhrase = phrases[phraseIndex]
    81→    let timeout: NodeJS.Timeout
    82→
    83→    if (!isDeleting && displayed === currentPhrase) {
    84→      timeout = setTimeout(() => setIsDeleting(true), pauseDuration)
    85→    } else if (isDeleting && displayed === '') {
    86→      setIsDeleting(false)
    87→      setPhraseIndex((prev) => (prev + 1) % phrases.length)
    88→    } else {
    89→      timeout = setTimeout(
    90→        () => {
    91→          setDisplayed(
    92→            isDeleting
    93→              ? currentPhrase.substring(0, displayed.length - 1)
    94→              : currentPhrase.substring(0, displayed.length + 1)
    95→          )
    96→        },
    97→        isDeleting ? deletingSpeed : typingSpeed
    98→      )
    99→    }
   100→    return () => clearTimeout(timeout)
   101→  }, [displayed, isDeleting, phraseIndex, phrases, typingSpeed, deletingSpeed, pauseDuration])
   102→
   103→  return displayed
   104→}

   106→/* ─── useLocalStorage Hook ─── */
   107→function useLocalStorage<T>(key: string, initialValue: T): [T, (v: T | ((prev: T) => T)) => void] {
   108→  const [stored, setStored] = useState<T>(() => {
   109→    if (typeof window === 'undefined') return initialValue
   110→    try {
   111→      const item = localStorage.getItem(key)
   112→      return item ? JSON.parse(item) : initialValue
   113→    } catch { return initialValue }
   114→  })
   115→  const setValue = useCallback((value: T | ((prev: T) => T)) => {
   116→    setStored(prev => {
   117→      const next = value instanceof Function ? value(prev) : value
   118→      if (typeof window !== 'undefined') localStorage.setItem(key, JSON.stringify(next))
   119→      return next
   120→    })
   121→  }, [key])
   122→  return [stored, setValue]
   123→}

   125→/* ─── useCountdown Hook ─── */
   126→function useCountdown(targetDate: Date) {
   127→  const [timeLeft, setTimeLeft] = useState({ days: 0, hours: 0, minutes: 0, seconds: 0 })
   128→  useEffect(() => {
   129→    const calc = () => {
   130→      const diff = targetDate.getTime() - Date.now()
   131→      if (diff <= 0) return { days: 0, hours: 0, minutes: 0, seconds: 0 }
   132→      return {
   133→        days: Math.floor(diff / 86400000),
   134→        hours: Math.floor((diff % 86400000) / 3600000),
   135→        minutes: Math.floor((diff % 3600000) / 60000),
   136→        seconds: Math.floor((diff % 60000) / 1000),
   137→      }
   138→    }
   139→    setTimeLeft(calc())
   140→    const interval = setInterval(() => setTimeLeft(calc()), 1000)
   141→    return () => clearInterval(interval)
   142→  }, [targetDate])
   143→  return timeLeft
   144→}

   146→/* ─── Animated Counter Component ─── */
   147→function AnimatedStat({ end, suffix, children }: { end: number; suffix?: string; children: React.ReactNode }) {
   148→  const [count, setCount] = useState(0)
   149→  const [started, setStarted] = useState(false)
   150→  const ref = useRef<HTMLDivElement>(null)
   151→
   152→  useEffect(() => {
   153→    if (!started) return
   154→    let start = 0
   155→    let frameId: number
   156→    const step = (timestamp: number) => {
   157→      if (!start) start = timestamp
   158→      const progress = Math.min((timestamp - start) / 1500, 1)
   159→      setCount(Math.floor(progress * end))
   160→      if (progress < 1) frameId = requestAnimationFrame(step)
   161→    }
   162→    frameId = requestAnimationFrame(step)
   163→    return () => cancelAnimationFrame(frameId)
   164→  }, [started, end])
   165→
   166→  useEffect(() => {
   167→    const el = ref.current
   168→    if (!el) return
   169→    const observer = new IntersectionObserver(
   170→      ([entry]) => { if (entry.isIntersecting) { setStarted(true); observer.disconnect() } },
   171→      { threshold: 0.3 }
   172→    )
   173→    observer.observe(el)
   174→    return () => observer.disconnect()
   175→  }, [])
   176→
   177→  return (
   178→    <div
   179→      ref={ref}
   180→      className="text-center"
   181→    >
   182→      {children}
   183→      <div className="text-3xl sm:text-4xl font-extrabold text-white mb-1">{count}{suffix}</div>
   184→    </div>
   185→  )
   186→}

   188→/* ─── Before / After Slider ─── */
   189→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   190→  const containerRef = useRef<HTMLDivElement>(null)
   191→  const [sliderPos, setSliderPos] = useState(50)
   192→  const [isDragging, setIsDragging] = useState(false)
   193→
   194→  const updatePosition = (clientX: number) => {
   195→    if (!containerRef.current) return
   196→    const rect = containerRef.current.getBoundingClientRect()
   197→    const x = clientX - rect.left
   198→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   199→    setSliderPos(pct)
   200→  }
   201→
   202→  const handlePointerDown = (e: React.PointerEvent) => {
   203→    setIsDragging(true)
   204→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   205→    updatePosition(e.clientX)
   206→  }
   207→
   208→  const handlePointerMove = (e: React.PointerEvent) => {
   209→    if (!isDragging) return
   210→    updatePosition(e.clientX)
   211→  }
   212→
   213→  const handlePointerUp = () => {
   214→    setIsDragging(false)
   215→  }
   216→
   217→  return (
   218→    <div className="space-y-2">
   219→      <div
   220→        ref={containerRef}
   221→        className="relative h-56 sm:h-64 rounded-xl overflow-hidden cursor-col-resize select-none border-2 border-border shadow-lg"
   222→        onPointerDown={handlePointerDown}
   223→        onPointerMove={handlePointerMove}
   224→        onPointerUp={handlePointerUp}
   225→      >
   226→        <img src={after} alt="Après" className="absolute inset-0 w-full h-full object-cover" draggable={false} />
   227→        <div className="absolute inset-0" style={{ clipPath: `inset(0 ${100 - sliderPos}% 0 0)` }}>
   228→          <img src={before} alt="Avant" className="w-full h-full object-cover" draggable={false} />
   229→        </div>
   230→        <span className="absolute top-3 left-3 z-20 bg-red-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">AVANT</span>
   231→        <span className="absolute top-3 right-3 z-20 bg-emerald-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">APRÈS</span>
   232→        <div className="absolute top-0 bottom-0 z-30 w-0.5 bg-white shadow-[0_0_8px_rgba(0,0,0,0.5)]" style={{ left: `${sliderPos}%` }}>
   233→          <div className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 w-10 h-10 rounded-full bg-white shadow-xl flex items-center justify-center border-2 border-white/80">
   234→            <div className="flex items-center gap-0.5">
   235→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-r-[6px] border-t-transparent border-b-transparent border-r-gray-700" />
   236→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-l-[6px] border-t-transparent border-b-transparent border-l-gray-700" />
   237→            </div>
   238→          </div>
   239→        </div>
   240→        {!isDragging && (
   241→          <div className="absolute inset-0 z-10 flex items-center justify-center opacity-0 hover:opacity-100 transition-opacity bg-black/20">
   242→            <div className="bg-white/90 text-gray-800 text-xs font-semibold px-3 py-1.5 rounded-full shadow">
   243→              Glissez pour comparer
   244→            </div>
   245→          </div>
   246→        )}
   247→      </div>
   248→    </div>
   249→  )
   250→}
