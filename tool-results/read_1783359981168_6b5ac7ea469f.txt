    70→/* ─── Animated Counter Component ─── */
    71→function AnimatedStat({ end, suffix, children }: { end: number; suffix?: string; children: React.ReactNode }) {
    72→  const [count, setCount] = useState(0)
    73→  const [started, setStarted] = useState(false)
    74→
    75→  useEffect(() => {
    76→    if (!started) return
    77→    let start = 0
    78→    let frameId: number
    79→    const step = (timestamp: number) => {
    80→      if (!start) start = timestamp
    81→      const progress = Math.min((timestamp - start) / 1500, 1)
    82→      setCount(Math.floor(progress * end))
    83→      if (progress < 1) frameId = requestAnimationFrame(step)
    84→    }
    85→    frameId = requestAnimationFrame(step)
    86→    return () => cancelAnimationFrame(frameId)
    87→  }, [started, end])
    88→
    89→  return (
    90→    <div
    91→      className="text-center"
    92→      onPointerEnter={() => setStarted(true)}
    93→    >
    94→      {children}
    95→      <div className="text-3xl sm:text-4xl font-extrabold text-white mb-1">{count}{suffix}</div>
    96→    </div>
    97→  )
    98→}

   100→/* ─── Before / After Slider ─── */
   101→function BeforeAfterSlider({ before, after }: { before: string; after: string }) {
   102→  const containerRef = useRef<HTMLDivElement>(null)
   103→  const [sliderPos, setSliderPos] = useState(50)
   104→  const [isDragging, setIsDragging] = useState(false)
   105→
   106→  const updatePosition = (clientX: number) => {
   107→    if (!containerRef.current) return
   108→    const rect = containerRef.current.getBoundingClientRect()
   109→    const x = clientX - rect.left
   110→    const pct = Math.max(0, Math.min(100, (x / rect.width) * 100))
   111→    setSliderPos(pct)
   112→  }
   113→
   114→  const handlePointerDown = (e: React.PointerEvent) => {
   115→    setIsDragging(true)
   116→    ;(e.target as HTMLElement).setPointerCapture(e.pointerId)
   117→    updatePosition(e.clientX)
   118→  }
   119→
   120→  const handlePointerMove = (e: React.PointerEvent) => {
   121→    if (!isDragging) return
   122→    updatePosition(e.clientX)
   123→  }
   124→
   125→  const handlePointerUp = () => {
   126→    setIsDragging(false)
   127→  }
   128→
   129→  return (
   130→    <div className="space-y-2">
   131→      <div
   132→        ref={containerRef}
   133→        className="relative h-56 sm:h-64 rounded-xl overflow-hidden cursor-col-resize select-none border-2 border-border shadow-lg"
   134→        onPointerDown={handlePointerDown}
   135→        onPointerMove={handlePointerMove}
   136→        onPointerUp={handlePointerUp}
   137→      >
   138→        <img src={after} alt="Après" className="absolute inset-0 w-full h-full object-cover" draggable={false} />
   139→        <div className="absolute inset-0" style={{ clipPath: `inset(0 ${100 - sliderPos}% 0 0)` }}>
   140→          <img src={before} alt="Avant" className="w-full h-full object-cover" draggable={false} />
   141→        </div>
   142→        <span className="absolute top-3 left-3 z-20 bg-red-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">AVANT</span>
   143→        <span className="absolute top-3 right-3 z-20 bg-emerald-500/90 text-white text-[10px] font-bold px-2.5 py-1 rounded-full backdrop-blur-sm">APRÈS</span>
   144→        <div className="absolute top-0 bottom-0 z-30 w-0.5 bg-white shadow-[0_0_8px_rgba(0,0,0,0.5)]" style={{ left: `${sliderPos}%` }}>
   145→          <div className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 w-10 h-10 rounded-full bg-white shadow-xl flex items-center justify-center border-2 border-white/80">
   146→            <div className="flex items-center gap-0.5">
   147→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-r-[6px] border-t-transparent border-b-transparent border-r-gray-700" />
   148→              <div className="w-0 h-0 border-t-[5px] border-b-[5px] border-l-[6px] border-t-transparent border-b-transparent border-l-gray-700" />
   149→            </div>
   150→          </div>
   151→        </div>
   152→        {!isDragging && (
   153→          <div className="absolute inset-0 z-10 flex items-center justify-center opacity-0 hover:opacity-100 transition-opacity bg-black/20">
   154→            <div className="bg-white/90 text-gray-800 text-xs font-semibold px-3 py-1.5 rounded-full shadow">
   155→              Glissez pour comparer
   156→            </div>
   157→          </div>
   158→        )}
   159→      </div>
   160→    </div>
   161→  )
   162→}
